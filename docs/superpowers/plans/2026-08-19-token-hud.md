# TokenHUD 实现计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 菜单栏 + 置顶悬浮窗 App，插件式显示多家 AI 供应商 token 余额，一键写入 Claude Code HUD。

**Architecture:** 主框架扫描两处插件目录（bundle 内置 + 用户 Application Support），spawn 每家 `fetch.sh` 读 stdout 用量 JSON，写 `~/.claude/hud/` 共享缓存（与 statusline 互通），SwiftUI 渲染菜单栏/悬浮窗/设置窗。

**Tech Stack:** Swift 6.1 / SwiftUI / MenuBarExtra + NSPanel / SwiftPM + XCTest / zsh 脚本插件 / python3（one-api 模板合并）

## Global Constraints

- 平台下限 macOS 14（`Package.swift` `.macOS(.v14)`）
- App 无 Dock 图标：Info.plist `LSUIElement=true`
- 零第三方依赖（Package.swift dependencies 为空）
- 缓存契约：`~/.claude/hud/.{cacheKey}-usage-cache.json`，字段 `generatedAt`(ms)/`five_hour`/`seven_day`/`monthly`/`error`/`stale`，桶内 `used_percentage`/`resets_at`(秒)
- 状态色阈值：used% ≥90 红、≥70 黄、其余绿；stale 加 `*`；`error=="auth"` 显示"需登录"
- 插件 slug 约束：`[a-z0-9-]`
- node 绝对路径由框架探测，spawn 时注入 env `TOKENHUD_NODE`
- 内置 4 家 manifest 的 `cacheFile` 必须指向现有缓存路径（glm/gpt/ollama/.oc-go-usage-cache.json），与 statusline 共享去重
- 每个任务结束 commit；提交信息格式 `feat|test|docs|chore: <desc>`
- App 永不 crash：单家失败隔离（stale 标记），顶层不因数据错误抛出

```swift
import Foundation

struct ProviderManifest: Codable {
    let name: String
    let label: String
    var fetch: String
    var intervalSec: Int?
    var buckets: [String: String]
    var cacheFile: String?
    var enabled: Bool?

    var isEnabled: Bool { enabled ?? true }
    var interval: Int { intervalSec ?? 60 }
}

struct Bucket: Codable {
    var usedPercentage: Double?
    var resetsAt: Int?

    enum CodingKeys: String, CodingKey {
        case usedPercentage = "used_percentage"
        case resetsAt = "resets_at"
    }
}

struct UsageCache: Codable {
    var generatedAt: Double?
    var fiveHour: Bucket?
    var sevenDay: Bucket?
    var monthly: Bucket?
    var error: String?
    var stale: Bool?

    enum CodingKeys: String, CodingKey {
        case generatedAt
        case fiveHour = "five_hour"
        case sevenDay = "seven_day"
        case monthly
        case error, stale
    }

    /// 解析 fetch.sh stdout（可缺 generatedAt/ttlMs 等字段），统一盖上框架字段
    static func make(body: Data, now: Double) -> UsageCache? {
        guard let raw = try? JSONDecoder().decode(UsageCache.self, from: body) else { return nil }
        var c = raw
        c.generatedAt = now
        c.stale = false
        if c.error == nil, c.fiveHour == nil, c.sevenDay == nil, c.monthly == nil { return nil }
        return c
    }
}

/// 一个已装载的插件（manifest + 所在目录）
struct PluginProvider {
    let slug: String
    let manifest: ProviderManifest
    let dir: URL
}
```

```swift
import Foundation

enum ProviderRegistry {
    static func isValidSlug(_ s: String) -> Bool {
        !s.isEmpty && s.allSatisfy { $0.isLowercase && $0.isNumber || $0 == "-" }
    }

    /// 扫单个插件根目录: <dir>/<slug>/manifest.json
    static func scan(dir: URL) -> [PluginProvider] {
        let fm = FileManager.default
        guard let entries = try? fm.contentsOfDirectory(at: dir, includingPropertiesForKeys: nil) else { return [] }
        return entries
            .filter { $0.hasDirectoryPath && isValidSlug($0.lastPathComponent) }
            .compactMap { slugDir -> PluginProvider? in
                let mfile = slugDir.appendingPathComponent("manifest.json")
                guard let data = try? Data(contentsOf: mfile),
                      let m = try? JSONDecoder().decode(ProviderManifest.self, from: data)
                else { return nil }
                return PluginProvider(slug: slugDir.lastPathComponent, manifest: m, dir: slugDir)
            }
            .sorted { $0.slug < $1.slug }
    }

    /// 双源合并: bundle 内置 + 用户目录(同名覆盖), 返回错误清单 (slug -> 原因)
    static func load(bundleDir: URL?, userDir: URL) -> (plugins: [PluginProvider], errors: [String: String]) {
        var errors: [String: String] = [:]
        let fm = FileManager.default

        // 用户目录里 manifest 损坏的也要报出来(不能静默)
        if let entries = try? fm.contentsOfDirectory(at: userDir, includingPropertiesForKeys: nil) {
            for e in entries where e.hasDirectoryPath && isValidSlug(e.lastPathComponent) {
                let mfile = e.appendingPathComponent("manifest.json")
                guard let data = try? Data(contentsOf: mfile),
                      (try? JSONDecoder().decode(ProviderManifest.self, from: data)) != nil
                else { errors[e.lastPathComponent] = "manifest.json 解析失败或缺失" }
            }
        }

        var merged: [String: PluginProvider] = [:]
        for p in (bundleDir.map(scan(dir:)) ?? []) { merged[p.slug] = p }
        for p in scan(dir: userDir) { merged[p.slug] = p }
        return (Array(merged.values).sorted { $0.slug < $1.slug }, errors)
    }
}
```

```swift
import Foundation

enum NodeLocator {
    /// nvmHome: ~/.nvm; saved: UserDefaults 存的路径; extraPaths: 兜底候选
    static func find(nvmHome: String?, extraPaths: [String], saved: String?) -> String? {
        if let saved, FileManager.default.fileExists(atPath: saved) { return saved }
        if let nvmHome, let p = pickLatest(from: URL(fileURLWithPath: nvmHome)) { return p }
        return extraPaths.first { FileManager.default.fileExists(atPath: $0) }
    }

    /// glob <nvm>/versions/node/*/bin/node, 取版本号最大 (语义化版本数值比较)
    static func pickLatest(from nvm: URL) -> String? {
        let base = nvm.appendingPathComponent("versions/node")
        guard let dirs = try? FileManager.default.contentsOfDirectory(at: base, includingPropertiesForKeys: nil) else { return nil }
        var best: (version: [Int], path: String)?
        for d in dirs {
            let node = d.appendingPathComponent("bin/node")
            guard FileManager.default.fileExists(atPath: node.path) else { continue }
            let parts = d.lastPathComponent
                .dropFirst(if: $0 == "v").split(separator: ".").compactMap { Int($0) }
            guard !parts.isEmpty else { continue }
            if best == nil || parts.lexicographicallyPrecedes(best!.version) == false && parts != best!.version || (parts > best!.version) {
                // 直接用数组比较
            }
            if best == nil || parts > best!.version { best = (parts, node.path) }
        }
        return best?.path
    }
}

final class ProviderRunner {
    let nodePath: String?

    init(nodePath: String?) { self.nodePath = nodePath }

    /// spawn: zsh -c 'exec ./fetch.sh' (cwd=插件目录, env 注入 TOKENHUD_NODE)
    /// 成功: stdout -> UsageCache.make; 任何失败(非0退出/解析失败/超时) -> nil
    func run(_ plugin: PluginProvider, timeout: TimeInterval = 15) async -> UsageCache? {
        let script = plugin.dir.appendingPathComponent(plugin.manifest.fetch).standardizedFileURL.path
        guard FileManager.default.fileExists(atPath: script) else { return nil }

        let p = Process()
        p.executableURL = URL(fileURLWithPath: "/bin/zsh")
        p.arguments = ["-c", "exec \"$1\"", script]
        p.currentDirectoryURL = plugin.dir
        var env = ProcessInfo.processInfo.environment
        if let nodePath { env["TOKENHUD_NODE"] = nodePath }
        p.environment = env

        let out = Pipe()
        p.standardOutput = out
        p.standardError = Pipe() // 丢弃

        do { try p.run() } catch { return nil }

        let deadline = Date().addingTimeInterval(timeout)
        while p.isRunning && Date() < deadline {
            try? await Task.sleep(nanoseconds: 100_000_000)
        }
        if p.isRunning {
            p.terminate()
            _ = p.waitUntilExit()
            return nil
        }
        guard p.terminationStatus == 0 else { return nil }

        let data = out.fileHandleForReading.readDataToEndOfFile()
        return UsageCache.make(body: data, now: Date().timeIntervalSince1970 * 1000)
    }
}
```

```swift
import Foundation
import SwiftUI

// MARK: - 纯函数

enum HUDFmt {
    enum Color { case red, yellow, green, dim }

    static func pctColor(_ p: Double?) -> Color {
        guard let p else { return .dim }
        if p >= 90 { return .red }
        if p >= 70 { return .yellow }
        return .green
    }

    static func resetCountdown(_ epochSec: Int?, now: Date) -> String? {
        guard let epochSec else { return nil }
        let diff = TimeInterval(epochSec) - now.timeIntervalSince1970
        guard diff > 0 else { return nil }
        let totalMin = Int(diff / 60)
        let h = totalMin / 60, m = totalMin % 60, d = h / 24
        return d > 0 ? "\(d)d\(h % 24)h" : "\(h)h\(m)m"
    }

    static func isFresh(_ cache: UsageCache?, interval: Int, now: Double) -> Bool {
        guard let g = cache?.generatedAt else { return false }
        return now - g < Double(interval) * 1000
    }
}

// MARK: - 状态

struct ProviderState: Identifiable {
    let slug: String
    let manifest: ProviderManifest
    var cache: UsageCache?
    var isStale: Bool
    var id: String { slug }

    var worstPct: Double? {
        [cache?.fiveHour?.usedPercentage, cache?.sevenDay?.usedPercentage, cache?.monthly?.usedPercentage]
            .compactMap { $0 }.max()
    }
}

// MARK: - Engine

@MainActor
final class DataEngine: ObservableObject {
    @Published private(set) var states: [ProviderState] = []
    @Published private(set) var registryErrors: [String: String] = [:]
    @Published private(set) var nodeMissing = false
    @Published private(set) var lastRefresh: Date?

    private let hudDir: URL
    private let bundleDir: URL?
    private let userDir: URL
    private let runner: ProviderRunner
    private let defaults: UserDefaults
    private var timer: Timer?

    static let defaultsNodeKey = "tokenhud.nodePath"

    init(hudDir: URL, bundleProvidersDir: URL?, userProvidersDir: URL,
         runner: ProviderRunner, defaults: UserDefaults = .standard) {
        self.hudDir = hudDir
        self.bundleDir = bundleProvidersDir
        self.userDir = userProvidersDir
        self.runner = runner
        self.defaults = defaults
    }

    /// cacheFile 展开: ~ 前缀 → HOME; 缺省 hudDir/.{slug}-usage-cache.json
    nonisolated static func cachePath(for plugin: PluginProvider, hudDir: URL) -> URL {
        if let cf = plugin.manifest.cacheFile {
            if cf.hasPrefix("~") {
                return URL(fileURLWithPath: String(cf.dropFirst())).path == ""
                    ? hudDir : URL(fileURLWithPath: NSHomeDirectory() + String(cf.dropFirst()))
            }
            return URL(fileURLWithPath: cf)
        }
        return hudDir.appendingPathComponent(".\(plugin.slug)-usage-cache.json")
    }

    func start() {
        reloadRegistry()
        timer = Timer.scheduledTimer(withTimeInterval: 10, repeats: true) { [weak self] _ in
            Task { @MainActor in await self?.refreshNow() }
        }
        Task { await refreshNow() }
    }

    func reloadRegistry() {
        let (plugins, errors) = ProviderRegistry.load(bundleDir: bundleDir, userDir: userDir)
        registryErrors = errors
        states = plugins.filter(\.manifest.isEnabled).map {
            ProviderState(slug: $0.slug, manifest: $0.manifest,
                          cache: readCache(slug: $0.slug, manifest: $0.manifest),
                          isStale: true)
        }
    }

    func refreshNow() async {
        let now = Date().timeIntervalSince1970 * 1000
        // 并行刷过期家
        await withTaskGroup(of: (String, UsageCache?).self) { group in
            for st in states {
                let plugin = PluginProvider(slug: st.slug, manifest: st.manifest,
                                            dir: pluginDir(slug: st.slug))
                guard !HUDFmt.isFresh(st.cache, interval: st.manifest.interval, now: now) else { continue }
                group.addTask { [runner] in
                    let c = await runner.run(plugin)
                    return (st.slug, c)
                }
            }
            for await (slug, cache) in group {
                apply(slug: slug, fetched: cache, now: now)
            }
        }
        lastRefresh = Date()
    }

    private func pluginDir(slug: String) -> URL {
        // 用户目录优先, 其次 bundle
        let u = userDir.appendingPathComponent(slug, isDirectory: true)
        if FileManager.default.fileExists(atPath: u.path) { return u }
        return bundleDir?.appendingPathComponent(slug, isDirectory: true) ?? u
    }

    private func apply(slug: String, fetched: UsageCache?, now: Double) {
        guard let i = states.firstIndex(where: { $0.slug == slug }) else { return }
        if let fetched {
            writeCache(fetched, plugin: states[i])
            states[i].cache = fetched
            states[i].isStale = false
        } else {
            // 失败: 保留旧值, 标 stale (旧值写盘也盖 stale 标记, 与 statusline 约定一致)
            var old = states[i].cache ?? UsageCache(generatedAt: now, fiveHour: nil, sevenDay: nil, monthly: nil, error: "fetch", stale: true)
            old.stale = true
            if old.error == nil { old.error = "fetch" }
            states[i].cache = old
            states[i].isStale = true
            writeCache(old, plugin: states[i])
        }
    }

    private func readCache(slug: String, manifest: ProviderManifest) -> UsageCache? {
        let p = DataEngine.cachePath(for: PluginProvider(slug: slug, manifest: manifest, dir: hudDir), hudDir: hudDir)
        guard let data = try? Data(contentsOf: p) else { return nil }
        return try? JSONDecoder().decode(UsageCache.self, from: data)
    }

    private func writeCache(_ cache: UsageCache, plugin state: ProviderState) {
        let p = DataEngine.cachePath(for: PluginProvider(slug: state.slug, manifest: state.manifest, dir: hudDir), hudDir: hudDir)
        if let data = try? JSONEncoder().encode(cache) {
            try? data.write(to: p, options: .atomic)
        }
    }
}
```

```swift
import Foundation

struct EnvsEntry: Codable {
    var label: String
    var source: String
    var cache: String
    var buckets: [String: String]
}

enum HUDSync {
    /// statusline ENV_REGISTRY 内置 key — envs.json 永不动它们
    static let builtinKeys: Set<String> = ["Claude-Max", "CC-GLM", "GPT", "Ollama", "OC-Go"]

    static func envsDict(plugins: [PluginProvider], hudDir: URL) -> [String: EnvsEntry] {
        var d: [String: EnvsEntry] = [:]
        for p in plugins where p.manifest.isEnabled && !builtinKeys.contains(p.manifest.name) {
            d[p.manifest.name] = EnvsEntry(
                label: p.manifest.label,
                source: "cache",
                cache: DataEngine.cachePath(for: p, hudDir: hudDir).path,
                buckets: p.manifest.buckets)
        }
        return d
    }

    /// new 全量替换自定义段; existing 里属于自定义(非内置 key)但 new 已不含的 → 删除
    static func mergeEnvs(existing: [String: EnvsEntry], new: [String: EnvsEntry]) -> [String: EnvsEntry] {
        var merged = existing.filter { builtinKeys.contains($0.key) }
        for (k, v) in new { merged[k] = v }
        return merged
    }

    /// ACCOUNT_ENVS: 原顺序保留仍然合法的 key, 新自定义 key 追加尾部
    static func mergeAccountEnvs(current: String?, keys: [String]) -> String {
        let currentList = (current ?? "").split(separator: ",").map(String.init).filter { !$0.isEmpty }
        let keep = currentList.filter { keys.contains($0) || builtinKeys.contains($0) }
        let append = keys.filter { !keep.contains($0) }
        return (keep + append).joined(separator: ",")
    }

    static func writeEnvs(_ dict: [String: EnvsEntry], to url: URL) throws {
        let enc = JSONEncoder()
        enc.outputFormatting = [.prettyPrinted, .sortedKeys]
        try enc.encode(dict).write(to: url, options: .atomic)
    }

    /// 只改 settings.json 的 env.ACCOUNT_ENVS, 其余字段原样保留 (JSONSerialization 全量)
    static func patchSettings(settingsURL: URL, accountEnvs: String) throws {
        let data = try Data(contentsOf: settingsURL)
        let obj = try JSONSerialization.jsonObject(with: data, options: [.mutateContainers]) as? [String: Any]
        guard var obj else { throw NSError(domain: "HUDSync", code: 1,
                                           userInfo: [NSLocalizedDescriptionKey: "settings.json 顶层非对象"]) }
        var env = obj["env"] as? [String: Any] ?? [:]
        env["ACCOUNT_ENVS"] = accountEnvs
        obj["env"] = env
        let out = JSONSerialization.data(withJSONObject: obj, options: [.prettyPrinted, .sortedKeys])
        try out.write(to: settingsURL, options: .atomic)
    }
```

```swift
import Foundation

enum PluginGenerator {
    static func slugify(_ name: String) -> String {
        let s = name.lowercased()
            .replacingOccurrences(of: " ", with: "-")
            .replacingOccurrences(of: "_", with: "-")
        let filtered = String(s.map { ch -> Character in
            (ch.isLowercase && ch.isLetter) || ch.isNumber || ch == "-" ? ch : "-"
        })
        let trimmed = filtered.filter { !$0.isWhitespace }
        return trimmed.isEmpty ? "custom" : trimmed
    }

    static func oneAPIManifest(label: String, intervalSec: Int = 60) -> ProviderManifest {
        ProviderManifest(name: label, label: label, fetch: "./fetch.sh",
                         intervalSec: intervalSec, buckets: ["seven_day": "周"],
                         cacheFile: nil, enabled: true)
    }

    static func oneAPIFetchScript(baseURL: String, apiKey: String) -> String {
        // baseURL 去尾斜杠; key 内嵌明文 (与 .glm-usage-auth.json 同级安全)
        let base = baseURL.hasSuffix("/") ? String(baseURL.dropLast()) : baseURL
        return """
        #!/bin/zsh
        # TokenHUD 生成: one-api / OpenAI 兼容模板
        set -euo pipeout
        BASE='\(base)'
        KEY='\(apiKey)'
        AUTH="Authorization: Bearer $KEY"
        START=$(date +%Y-%m-01)
        END=$(date -v+1d +%Y-%m-%d)
        SUB=$(curl -s --max-time 10 -H "$AUTH" "$BASE/v1/dashboard/billing/subscription")
        USAGE=$(curl -s --max-time 10 -H "$AUTH" "$BASE/v1/dashboard/billing/usage?start_date=$START&end_date=$END")
        print -r -- "$SUB\\n$USAGE" | python3 -c '
        import json, sys
        lines = sys.stdin.read().split("\\n")
        sub = json.loads(lines[0])
        usage = json.loads(lines[1])
        quota = sub.get("hard_quota") or sub.get("total_granted") or 0
        total = usage.get("total_usage") or 0
        if not quota or quota <= 0:
            print(json.dumps({"error": "quota"})); raise SystemExit(1)
        pct = round(total / quota * 100, 1)
        print(json.dumps({"seven_day": {"used_percentage": pct}}))
        '
        """
    }

    static func writePlugin(slug: String, manifest: ProviderManifest, script: String, into root: URL) throws {
        let dir = root.appendingPathComponent(slug, isDirectory: true)
        try FileManager.default.createDirectory(at: dir, withIntermediateDirectories: true)
        let enc = JSONEncoder()
        enc.outputFormatting = [.prettyPrinted, .sortedKeys]
        try enc.encode(manifest).write(to: dir.appendingPathComponent("manifest.json"), options: .atomic)
        try script.write(to: dir.appendingPathComponent("fetch.sh"), atomically: true, encoding: .utf8)
        try FileManager.default.setAttributes([.posixPermissions: 0o755],
                                              ofItemAtPath: dir.appendingPathComponent("fetch.sh").path)
    }

    static func removePlugin(slug: String, in root: URL) throws {
        try FileManager.default.removeItem(at: root.appendingPathComponent(slug, isDirectory: true))
    }
}
```

注意 `set -euo pipeout`：zsh 无 `pipefail` 关键字差异，此处意图是 curl 失败即退出——用 `set -eu` 即可，`-o pipeout` 是笔误，实现时写 `set -eu`。

```swift
        return s < 60 ? "\(s)s前" : "\(s / 60)m前"
    }
}

private struct MiniRow { let label: String; let bucket: Bucket }

extension FloatingView {
    func zipBucketRows(_ st: ProviderState) -> [MiniRow] {
        var out: [MiniRow] = []
        if let b = st.cache?.fiveHour, let l = st.manifest.buckets["five_hour"] { out.append(.init(label: l, bucket: b)) }
        if let b = st.cache?.sevenDay, let l = st.manifest.buckets["seven_day"] { out.append(.init(label: l, bucket: b)) }
        if let b = st.cache?.monthly, let l = st.manifest.buckets["monthly"] { out.append(.init(label: l, bucket: b)) }
        return out
    }
}
```

（`ForEach(..., id: \.label)` 要求 `MiniRow` 可识别——`MiniRow` 结构体以 `label` 为 id；`private struct` 移到与 `FloatingView` 同级可见处即可，实现时把 `private` 去掉。）

```swift
@State private var panelVisible = false
@State private var panelExpanded = false
private var panel: FloatingPanel?

private func togglePanel() {
    if let panel, panel.isVisible {
        panel.orderOut(nil)
        panelVisible = false
    } else {
        let hosting = NSHostingView(rootView: FloatingView(engine: engine, expanded: panelExpanded) {
            panelExpanded.toggle(); rebuildPanel()
        })
        let p = FloatingPanel(contentView: hosting)
        p.setContentSize(hosting.fittingSize)
        p.orderFrontRegardless()
        panel = p
        panelVisible = true
    }
}

private func rebuildPanel() {
    guard let panel else { return }
    let hosting = NSHostingView(rootView: FloatingView(engine: engine, expanded: panelExpanded) {
        panelExpanded.toggle(); rebuildPanel()
    })
    panel.contentView = hosting
    panel.setContentSize(hosting.fittingSize)
}
```

菜单栏 `onTogglePanel: { togglePanel() }`。`@State` 在 App struct 上不可变持有 NSPanel——实现时把 panel 管理移到一个 `final class PanelController: ObservableObject` 单例（`@StateObject` 持有），避免 App struct 值语义问题。

```swift
import SwiftUI
import AppKit

struct SettingsView: View {
    @ObservedObject var engine: DataEngine
    let hudDir: URL
    let userDir: URL
    let settingsURL: URL
    let bundleDir: URL?

    @State private var newName = ""
    @State private var newBase = ""
    @State private var newKey = ""
    @State private var testing = false
    @State private var testResult: String?
    @State private var syncResult: String?

    var body: some View {
        Form {
            Section("供应商") {
                ForEach(engine.states) { st in
                    HStack {
                        Text(st.manifest.label)
                        Text(st.slug).foregroundStyle(.secondary).font(.caption)
                        Spacer()
                        Text(st.cache?.error ?? formatPct(st.worstPct))
                            .foregroundStyle(hudColor(HUDFmt.pctColor(st.worstPct)))
                        if !isBuiltin(st.slug) {
                            Button("删除", role: .destructive) { delete(st.slug) }
                        }
                    }
                }
                ForEach(engine.registryErrors.sorted(by: { $0.key < $1.key }), id: \.key) { slug, msg in
                    HStack { Text(slug).foregroundStyle(.orange); Text(msg).font(.caption) }
                }
            }

            Section("添加 (one-api / OpenAI 兼容)") {
                TextField("名称 (如 my-relay)", text: $newName)
                TextField("baseURL (https://…)", text: $newBase)
                SecureField("apiKey", text: $newKey)
                HStack {
                    Button("测试连接") { test() }.disabled(newBase.isEmpty || newKey.isEmpty || testing)
                    if testing { ProgressView().controlSize(.small) }
                    if let testResult { Text(testResult).font(.caption) }
                }
                Button("保存") { save() }.disabled(newName.isEmpty || newBase.isEmpty || newKey.isEmpty)
            }

            Section("写入 Claude Code HUD") {
                Button("写入 Claude Code HUD") { syncHUD() }
                if let syncResult { Text(syncResult).font(.caption) }
            }

            Section("环境") {
                HStack {
                    Text("node: \(engine.nodeMissing ? "未找到" : "已定位")")
                    Button("手动指定…") { pickNode() }
                }
            }
        }
        .formStyle(.grouped)
        .frame(width: 460, height: 480)
    }

    private func isBuiltin(_ slug: String) -> Bool {
        ["glm", "gpt", "ollama", "opencode-go"].contains(slug)
    }

    private func test() {
        testing = true; testResult = nil
        let script = PluginGenerator.oneAPIFetchScript(baseURL: newBase, apiKey: newKey)
        Task {
            let tmp = FileManager.default.temporaryDirectory
                .appendingPathComponent("tokenhud-test-\(UUID())")
            try? FileManager.default.createDirectory(at: tmp, withIntermediateDirectories: true)
            try? script.write(to: tmp.appendingPathComponent("fetch.sh"), atomically: true, encoding: .utf8)
            let m = PluginGenerator.oneAPIManifest(label: newName.isEmpty ? "test" : newName)
            let runner = ProviderRunner(nodePath: NodeLocator.find(
                nvmHome: NSHomeDirectory() + "/.nvm",
                extraPaths: ["/opt/homebrew/bin/node", "/usr/local/bin/node"],
                saved: UserDefaults.standard.string(forKey: DataEngine.defaultsNodeKey)))
            let cache = await runner.run(PluginProvider(slug: "test", manifest: m, dir: tmp), timeout: 15)
            try? FileManager.default.removeItem(at: tmp)
            await MainActor.run {
                testing = false
                if let pct = cache?.sevenDay?.usedPercentage {
                    testResult = "✓ 已用 \(pct)%"
                } else {
                    testResult = "✗ 连接失败 (查 baseURL/key)"
                }
            }
        }
    }

    private func save() {
        let slug = PluginGenerator.slugify(newName)
        let m = PluginGenerator.oneAPIManifest(label: newName)
        let script = PluginGenerator.oneAPIFetchScript(baseURL: newBase, apiKey: newKey)
        do {
            try FileManager.default.createDirectory(at: userDir, withIntermediateDirectories: true)
            try PluginGenerator.writePlugin(slug: slug, manifest: m, script: script, into: userDir)
            newName = ""; newBase = ""; newKey = ""
            engine.reloadRegistry()
            Task { await engine.refreshNow() }
        } catch {
            testResult = "保存失败: \(error.localizedDescription)"
        }
    }

    private func delete(_ slug: String) {
        try? PluginGenerator.removePlugin(slug: slug, in: userDir)
        engine.reloadRegistry()
    }

    private func syncHUD() {
        do {
            let plugins = ProviderRegistry.load(bundleDir: bundleDir, userDir: userDir).plugins
            let new = HUDSync.envsDict(plugins: plugins, hudDir: hudDir)
            let envsURL = hudDir.appendingPathComponent("envs.json")
            let existing = (try? Data(contentsOf: envsURL))
                .flatMap { try? JSONDecoder().decode([String: EnvsEntry].self, from: $0) } ?? [:]
            try HUDSync.writeEnvs(HUDSync.mergeEnvs(existing: existing, new: new), to: envsURL)
            let keys = plugins.filter(\.manifest.isEnabled).map(\.manifest.name)
            let settings = try JSONSerialization.jsonObject(with: Data(contentsOf: settingsURL)) as? [String: Any]
            let current = (settings?["env"] as? [String: Any])?["ACCOUNT_ENVS"] as? String
            try HUDSync.patchSettings(settingsURL: settingsURL,
                accountEnvs: HUDSync.mergeAccountEnvs(current: current, keys: keys))
            syncResult = "✓ 已写入 (statusline 下次渲染生效)"
        } catch {
            syncResult = "✗ \(error.localizedDescription)"
        }
    }

    private func pickNode() {
        let panel = NSOpenPanel()
        panel.canChooseFiles = true; panel.canChooseDirectories = false
        panel.allowedContentTypes = [.data]
        if panel.runModal() == .OK, let url = panel.url {
            UserDefaults.standard.set(url.path, forKey: DataEngine.defaultsNodeKey)
            syncResult = "node 路径已存, 重启 App 生效"
        }
    }
}
```

```bash
#!/bin/zsh
# TokenHUD 打包: swift build → 拼 .app → ad-hoc 签名 → 装到 ~/Applications
set -euo pipefail
cd "$(dirname "$0")/.."
OUT=".build/TokenHUD.app"
rm -rf "$OUT"
swift build -c release
BIN=$(swift build -c release --show-bin-path)
APP="$OUT/Contents"
mkdir -p "$APP/MacOS" "$APP/Resources"
cp "$BIN/TokenHUD" "$APP/MacOS/"
cp -R "$BIN/TokenHUD_TokenHUD.bundle" "$APP/Resources/"
cat > "$APP/Info.plist" <<'EOF'
<?xml version="1.0" encoding="UTF-8"?>
<!DOCTYPE plist PUBLIC "-//Apple//DTD PLIST 1.0//EN" "http://www.apple.com/DTDs/PropertyList-1.0.dtd">
<plist version="1.0">
<dict>
    <key>CFBundleExecutable</key><string>TokenHUD</string>
    <key>CFBundleIdentifier</key><string>local.tokenhud</string>
    <key>CFBundleName</key><string>TokenHUD</string>
    <key>CFBundlePackageType</key><string>APPL</string>
    <key>CFBundleShortVersionString</key><string>0.1.0</string>
    <key>LSUIElement</key><true/>
    <key>NSHighResolutionCapable</key><true/>
</dict>
</plist>
EOF
printf 'APPL????' > "$APP/PkgInfo"
codesign --force --deep -s - "$OUT"
DEST="$HOME/Applications"
if pgrep -x TokenHUD >/dev/null; then
  echo "TokenHUD 正在运行, 先退出再安装 (菜单栏 → 退出)"; exit 1
fi
mkdir -p "$DEST"
cp -R "$OUT" "$DEST/"
echo "✓ 安装完成: $DEST/TokenHUD.app"
```
